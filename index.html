const SCRIPT_URL = "URL_WEB_APP_APPS_SCRIPT_ANDA_DI_SINI"; // Pastikan berakhiran /exec document.addEventListener('touchmove', function(e) { e.preventDefault(); }, { passive: false }); function prosesCari() { var hpInput = document.getElementById('noHP'); var hp = hpInput.value.trim(); var formSec = document.getElementById('formSection'); var resultDiv = document.getElementById('result'); var loadingDiv = document.getElementById('loading'); var errorDiv = document.getElementById('errorMessage'); resultDiv.style.display = 'none'; loadingDiv.style.display = 'none'; errorDiv.style.display = 'none'; if (!hp) { errorDiv.innerText = 'Silakan masukkan nomor HP terlebih dahulu.'; errorDiv.style.display = 'block'; return; } loadingDiv.style.display = 'block'; fetch(SCRIPT_URL, { method: 'POST', headers: { 'Content-Type': 'text/plain;charset=utf-8' }, body: JSON.stringify({ noHP: hp }), redirect: 'follow' }) .then(response => response.json()) .then(data => { loadingDiv.style.display = 'none'; if (data.success) { formSec.style.display = 'none'; document.getElementById('namaPeserta').innerText = data.nama; document.getElementById('nomorBIB').innerText = data.bib; resultDiv.style.display = 'block'; } else { errorDiv.innerText = data.message; errorDiv.style.display = 'block'; } }) .catch(error => { loadingDiv.style.display = 'none'; errorDiv.innerText = 'Gagal terhubung ke server. Pastikan URL Script benar.'; errorDiv.style.display = 'block'; }); }
